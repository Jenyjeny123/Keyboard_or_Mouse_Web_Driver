void ap_protocol_process(void) {
    if (g_recieve_flag == 0) return;
    g_recieve_flag = 0;
    
    // 打印接收到的缓冲区
    ap_protocol_print_buffer();
    
    // 检查是否为AP通讯数据（report id == 0x04）
    if (g_usb_recieve_buf[0] == AP_REPORT_ID) {
        // 处理AP通讯协议
        uint8_t cmd = g_usb_recieve_buf[4]; // 第五个字节：CMD
        
        switch (cmd) {
            case AP_CMD_READ_BASIC_INFO:
            {
                uint8_t length = g_usb_recieve_buf[5]; // 第六个字节：读取长度
                uint16_t offset = (g_usb_recieve_buf[7] << 8) | g_usb_recieve_buf[6]; // 第七、八字节：偏移地址（小端序）
                
                printf("AP Command: Read Basic Info\r\n");
                printf("Length: %d, Offset: %d\r\n", length, offset);
                
                // 准备响应数据
                uint8_t response[64] = {0};
                
                // 前8个字节与接收到的数据一致
                for (int i = 0; i < 8; i++) {
                    response[i] = g_usb_recieve_buf[i];
                }
                
                // 从第9个字节开始，填充基本信息区数据
                if (offset < HEAD_TAB_SIZE) {
                    uint8_t actual_length = (offset + length) > HEAD_TAB_SIZE ? (HEAD_TAB_SIZE - offset) : length;
                    for (int i = 0; i < actual_length; i++) {
                        response[8 + i] = head_description_tab[offset + i];
                    }
                }
                
                // 发送响应数据
                ap_protocol_send_response(response, 8 + length);
            }
            break;
            
            case AP_CMD_WRITE_BASIC_INFO:
            {
                printf("AP Command: Write Basic Info\r\n");
                // 后续实现
            }
            break;
            
            case AP_CMD_READ_RAM:
            {
                printf("AP Command: Read Function Area Params\r\n");
                // 后续实现
            }
            break;
            
            case AP_CMD_WRITE_RAM:
